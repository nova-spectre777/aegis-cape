cmake_minimum_required(VERSION 3.20)
project(aegis_cape LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(aegis_cape src/solver.cpp src/envelope.cpp)
target_include_directories(aegis_cape PUBLIC include)
add_executable(aegis-cape src/main.cpp)
target_link_libraries(aegis-cape PRIVATE aegis_cape)
add_executable(aegis-tests tests/test_solver.cpp)
target_link_libraries(aegis-tests PRIVATE aegis_cape)
enable_testing()
add_test(NAME aegis-tests COMMAND aegis-tests)
